                        if (!acc[task.category]) acc[task.category] = [];
                        acc[task.category].push(task);
                        return acc;
                    }, {});

                    return Object.keys(groupedTasks).sort().map(category => {
                        const categoryTotal = groupedTasks[category].reduce((sum, task) => {
                            const price = parseFloat(task.price) || 0;
                            const qty = parseFloat(task.quantity) || 0;
                            return sum + (price * qty);
                        }, 0);

                        return (
                            <div key={category} className="mb-8 bg-white rounded-2xl shadow-sm border border-slate-200 overflow-hidden transform transition-all duration-300 hover:shadow-md">
                                <div className={`px-5 py-4 border-b border-slate-100 font-bold ${getCategoryColor(category)} flex justify-between items-center`}>
                                    <div className="flex items-center gap-2">
                                        <svg className="w-5 h-5 opacity-70" xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><path d="M22 19a2 2 0 0 1-2 2H4a2 2 0 0 1-2-2V5a2 2 0 0 1 2-2h5l2 3h9a2 2 0 0 1 2 2z"></path></svg>
                                        <span className="text-base tracking-wide">{category}</span>
                                    </div>
                                    <span className="text-xs font-bold px-3 py-1 bg-white/60 rounded-full shadow-sm">{groupedTasks[category].length} Item</span>
                                </div>
                                <div className="overflow-x-auto">
                                    <table className="w-full text-left border-collapse min-w-[1050px]">
                                        <thead className="bg-slate-50/80 text-slate-600">
                                            <tr>
                                                <th className="px-3 py-3 border-b border-slate-200 font-bold text-xs w-12 text-center uppercase tracking-wider">Status</th>
                                                <th className="px-2 py-3 border-b border-slate-200 font-bold text-xs w-24 uppercase tracking-wider">Bulan</th>
                                                <th className="px-2 py-3 border-b border-slate-200 font-bold text-xs w-28 uppercase tracking-wider">Kategori</th>
                                                <th className="px-2 py-3 border-b border-slate-200 font-bold text-xs w-48 uppercase tracking-wider">Tugas / Item</th>
                                                <th className="px-2 py-3 border-b border-slate-200 font-bold text-xs w-32 uppercase tracking-wider">Harga Satuan</th>
                                                <th className="px-2 py-3 border-b border-slate-200 font-bold text-xs w-16 text-center uppercase tracking-wider">Jumlah</th>
                                                <th className="px-2 py-3 border-b border-slate-200 font-bold text-xs w-20 uppercase tracking-wider text-center">Satuan</th>
                                                <th className="px-3 py-3 border-b border-slate-200 font-bold text-xs w-32 text-right uppercase tracking-wider">Subtotal</th>
                                                <th className="px-2 py-3 border-b border-slate-200 font-bold text-xs uppercase tracking-wider">Catatan</th>
                                                <th className="px-2 py-3 border-b border-slate-200 font-bold text-xs w-12 text-center uppercase tracking-wider">Aksi</th>
                                            </tr>
                                        </thead>
                                        <tbody className="divide-y divide-slate-50">
                                            {groupedTasks[category].map((task, index) => renderTableRow(task, index))}
                                        </tbody>
                                        <tfoot className="bg-slate-50 border-t border-slate-200">
                                            <tr>
                                                <td colSpan="7" className="px-4 py-4 text-right font-bold text-slate-500 text-sm uppercase tracking-wider">
                                                    Total {category}:
                                                </td>
                                                <td className="px-3 py-4 text-right font-extrabold text-rose-600 text-base">
                                                    {formatIDR(categoryTotal)}
                                                </td>
                                                <td colSpan="2"></td>
                                            </tr>
                                        </tfoot>
                                    </table>
                                </div>
                            </div>
                        );
                    });
                }

                return (
                    <div className="bg-white rounded-2xl shadow-sm border border-slate-200 overflow-hidden transform transition-all hover:shadow-md">
                        <div className="overflow-x-auto max-h-[650px] overflow-y-auto relative styled-scrollbar">
                            <table className="w-full text-left border-collapse min-w-[1100px]">
                                <thead className="glass-header sticky top-0 z-10 shadow-sm bg-rose-50/90 text-slate-700">
                                    <tr>
                                        <th className="px-3 py-4 border-b border-rose-100 font-bold text-xs w-12 text-center uppercase tracking-wider">Status</th>
                                        <th className="px-2 py-4 border-b border-rose-100 font-bold text-xs w-24 uppercase tracking-wider">Bulan</th>
                                        <th className="px-2 py-4 border-b border-rose-100 font-bold text-xs w-28 uppercase tracking-wider">Kategori</th>
                                        <th className="px-2 py-4 border-b border-rose-100 font-bold text-xs w-48 uppercase tracking-wider">Tugas / Item</th>
                                        <th className="px-2 py-4 border-b border-rose-100 font-bold text-xs w-32 uppercase tracking-wider">Harga Satuan</th>
                                        <th className="px-2 py-4 border-b border-rose-100 font-bold text-xs w-16 text-center uppercase tracking-wider">Jumlah</th>
                                        <th className="px-2 py-4 border-b border-rose-100 font-bold text-xs w-20 text-center uppercase tracking-wider">Satuan</th>
                                        <th className="px-3 py-4 border-b border-rose-100 font-bold text-xs w-32 text-right uppercase tracking-wider">Subtotal</th>
                                        <th className="px-2 py-4 border-b border-rose-100 font-bold text-xs uppercase tracking-wider">Catatan</th>
                                        <th className="px-2 py-4 border-b border-rose-100 font-bold text-xs w-12 text-center uppercase tracking-wider">Aksi</th>
                                    </tr>
                                </thead>
                                <tbody className="divide-y divide-slate-50">
                                    {tasks.map((task, index) => renderTableRow(task, index))}
                                </tbody>
                            </table>
                        </div>
                    </div>
                );
            };

            const progress = tasks.length > 0 ? Math.round((tasks.filter(t => t.status).length / tasks.length) * 100) : 0;
            
            const grandTotalBudget = tasks.reduce((sum, task) => {
                const price = parseFloat(task.price) || 0;
                const qty = parseFloat(task.quantity) || 0;
                return sum + (price * qty);
            }, 0);

            return (
                <div className="max-w-7xl mx-auto py-4">
                    {/* Header Panel */}
                    <div className="bg-white rounded-2xl shadow-md border border-slate-100 p-6 mb-6 flex flex-col lg:flex-row justify-between items-start lg:items-center gap-6 relative overflow-hidden">
                        {/* Dekorasi Background */}
                        <div className="absolute top-0 right-0 -mr-16 -mt-16 w-48 h-48 bg-rose-50 rounded-full opacity-50 pointer-events-none"></div>
                        <div className="absolute bottom-0 left-0 -ml-16 -mb-16 w-32 h-32 bg-amber-50 rounded-full opacity-50 pointer-events-none"></div>

                        <div className="flex-1 w-full relative z-10">
                            <div className="flex items-center gap-3 mb-2">
                                <div className="p-2 bg-rose-100 text-rose-600 rounded-lg shadow-sm">
                                    <svg xmlns="http://www.w3.org/2000/svg" width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><path d="M14 2H6a2 2 0 0 0-2 2v16a2 2 0 0 0 2 2h12a2 2 0 0 0 2-2V8z"></path><polyline points="14 2 14 8 20 8"></polyline><line x1="16" y1="13" x2="8" y2="13"></line><line x1="16" y1="17" x2="8" y2="17"></line><polyline points="10 9 9 9 8 9"></polyline></svg>
                                </div>
                                <input 
                                    type="text"
                                    value={title}
                                    onChange={(e) => setTitle(e.target.value)}
                                    className="text-2xl sm:text-3xl font-extrabold text-slate-800 bg-transparent border-b-2 border-transparent hover:border-slate-200 focus:border-rose-400 focus:outline-none w-full max-w-lg transition-colors pb-1"
                                    title="Klik untuk mengubah judul"
                                />
                            </div>
                            <p className="text-slate-500 mt-1 ml-12 text-sm font-medium">Klik langsung pada teks, angka, atau dropdown di tabel untuk merencanakan mimpimu.</p>
                        </div>
                        
                        <div className="flex flex-col sm:flex-row items-center gap-4 w-full lg:w-auto relative z-10">
                            {/* Card Total Biaya */}
                            <div className="flex-1 sm:flex-none flex flex-col bg-slate-50/80 backdrop-blur-sm px-6 py-4 rounded-xl border border-slate-200/60 shadow-inner w-full min-w-[220px]">
                                <span className="text-xs font-bold text-slate-500 uppercase tracking-widest mb-1 flex items-center gap-1.5">
                                    <svg className="w-4 h-4 text-emerald-500" xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><line x1="12" y1="1" x2="12" y2="23"></line><path d="M17 5H9.5a3.5 3.5 0 0 0 0 7h5a3.5 3.5 0 0 1 0 7H6"></path></svg>
                                    Estimasi Biaya
                                </span>
                                <span className="text-2xl font-black text-slate-800 tracking-tight">{formatIDR(grandTotalBudget)}</span>
                            </div>

                            {/* Card Progres */}
                            <div className="flex-1 sm:flex-none flex flex-col bg-slate-50/80 backdrop-blur-sm px-6 py-4 rounded-xl border border-slate-200/60 shadow-inner w-full min-w-[220px]">
                                <div className="flex justify-between items-center mb-2">
                                    <span className="text-xs font-bold text-slate-500 uppercase tracking-widest flex items-center gap-1.5">
                                        <svg className="w-4 h-4 text-blue-500" xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><path d="M22 11.08V12a10 10 0 1 1-5.93-9.14"></path><polyline points="22 4 12 14.01 9 11.01"></polyline></svg>
                                        Progres
                                    </span>
                                    <span className="text-sm font-black text-slate-700">{progress}%</span>
                                </div>
                                <div className="w-full bg-slate-200 rounded-full h-2.5 shadow-inner overflow-hidden">
                                    <div className="bg-gradient-to-r from-rose-400 to-rose-600 h-2.5 rounded-full transition-all duration-700 ease-out" style={{ width: `${progress}%` }}></div>
                                </div>
                            </div>
                        </div>
                    </div>

                    {/* Toolbar Actions */}
                    <div className="flex flex-wrap gap-4 mb-6 justify-between items-center bg-white p-2 sm:p-3 rounded-xl shadow-sm border border-slate-200">
                        <div className="flex flex-wrap gap-3">
                            <button 
                                onClick={addTask}
                                className="bg-gradient-to-r from-rose-500 to-rose-600 hover:from-rose-600 hover:to-rose-700 text-white px-5 py-2.5 rounded-lg font-semibold text-sm transition-all shadow-sm hover:shadow-md transform hover:-translate-y-0.5 flex items-center gap-2"
                            >
                                <svg xmlns="http://www.w3.org/2000/svg" width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round"><line x1="12" y1="5" x2="12" y2="19"></line><line x1="5" y1="12" x2="19" y2="12"></line></svg>
                                Tambah Item Baru
                            </button>
                            
                            <label className="flex items-center gap-2.5 bg-slate-50 border border-slate-200 px-5 py-2.5 rounded-lg cursor-pointer hover:bg-slate-100 transition-all hover:shadow-sm">
                                <div className="relative flex items-center justify-center">
                                    <input 
                                        type="checkbox" 
                                        checked={groupByCategory} 
                                        onChange={() => setGroupByCategory(!groupByCategory)}
                                        className="w-5 h-5 text-rose-500 accent-rose-500 cursor-pointer peer"
                                    />
                                </div>
                                <span className="text-sm font-bold text-slate-700 select-none">Kelompokkan Per Kategori</span>
                            </label>
                        </div>

                        <div className="flex flex-wrap gap-3">
                            {/* Tombol Baru untuk Sync Google Sheets */}
                            <button 
                                onClick={syncToGoogleSheets}
                                disabled={syncStatus === 'syncing'}
                                className={`px-5 py-2.5 rounded-lg font-semibold text-sm transition-all shadow-sm flex items-center gap-2 border ${
                                    syncStatus === 'success' ? 'bg-emerald-50 text-emerald-700 border-emerald-300' :
                                    syncStatus === 'error' ? 'bg-red-50 text-red-700 border-red-300' :
                                    syncStatus === 'missing_url' ? 'bg-amber-50 text-amber-700 border-amber-300' :
                                    'bg-white hover:bg-blue-50 text-blue-600 border-blue-200 hover:border-blue-300 hover:shadow-md'
                                }`}
                            >
                                <svg xmlns="http://www.w3.org/2000/svg" width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
                                    {syncStatus === 'syncing' ? (
                                        <path d="M21 12a9 9 0 1 1-6.219-8.56" className="animate-spin" />
                                    ) : syncStatus === 'success' ? (
                                        <polyline points="20 6 9 17 4 12" />
                                    ) : syncStatus === 'missing_url' ? (
                                        <>
                                            <circle cx="12" cy="12" r="10"></circle>
                                            <line x1="12" y1="8" x2="12" y2="12"></line>
                                            <line x1="12" y1="16" x2="12.01" y2="16"></line>
                                        </>
                                    ) : (
                                        <>
                                            <path d="M21 15v4a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2v-4"></path>
                                            <polyline points="17 8 12 3 7 8"></polyline>
                                            <line x1="12" y1="3" x2="12" y2="15"></line>
                                        </>
                                    )}
                                </svg>
                                {syncStatus === 'syncing' ? 'Menyinkronkan...' :
                                 syncStatus === 'success' ? 'Berhasil Disimpan!' :
                                 syncStatus === 'error' ? 'Gagal Simpan' :
                                 syncStatus === 'missing_url' ? 'Isi URL Script Dulu' :
                                 'Sync ke Google Sheets'}
                            </button>

                            <button 
                                onClick={exportCSV}
                                className="bg-white hover:bg-slate-50 text-slate-700 border border-slate-300 px-5 py-2.5 rounded-lg font-semibold text-sm transition-all shadow-sm hover:shadow-md flex items-center gap-2"
                            >
                                <svg className="text-emerald-600" xmlns="http://www.w3.org/2000/svg" width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><path d="M21 15v4a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2v-4"></path><polyline points="7 10 12 15 17 10"></polyline><line x1="12" y1="15" x2="12" y2="3"></line></svg>
                                Export ke CSV
                            </button>
                        </div>
                    </div>

                    {/* Tabel Data */}
                    {renderContent()}

                    <div className="mt-8 text-center text-slate-400 text-sm pb-12 font-medium">
                        <p>Tips: Geser tabel ke kanan pada layar kecil untuk melihat kolom Harga, Jumlah, dan Subtotal.</p>
                    </div>
                </div>
            );
        };

        const root = ReactDOM.createRoot(document.getElementById('root'));
        root.render(<WeddingSpreadsheet />);
    </script>
